28 * @ingroup utils
29 * @brief all parser patterns in one struct
30 */
31 struct base_parser_patterns {
32 /// @brief pattern to get scalar data
33 static inline const auto scalarsPattern = pcre2cpp::regex(R"(^\s*+(?<scalar>(?>bool|double|float|u?int))\s*+$)");
34
35 /// @brief pattern to get vec data
36 static inline const auto vecPattern = pcre2cpp::regex(R"(^\s*+(?>(?<scalar>[idbuf])?vec(?<length>[2-4]))\s*+$)");
37
38 /// @brief pattern to get mat data
39 static inline const auto matPattern =
40 pcre2cpp::regex(R"(^\s*+(?>(?<scalar>[idbuf])?mat(?<cols>[2-4])(?>(?:x(?<rows>[2-4]))?))\s*+$)");
41
42 /// @brief pattern to get variable data
43 static inline const auto variablePattern = pcre2cpp::regex(
44 R"(^\s*+(?>(?<type>[a-zA-Z_]\w*+))\s++(?>(?<name>[a-zA-Z_]\w*+))\s*+(?>(?<array>\‍[\s*+(?<count>\d+)?\s*+\‍]))?\s*+(?>;?)\s*+$)"
45 );
46
47 /// @brief pattern to get multiple variables
48 static inline const auto multiVariablesPattern = pcre2cpp::regex(
49 R"((?>(?<=^|;))\s*+(?<var>[a-zA-Z_]\w*+\s++[a-zA-Z_]\w*+(?>(?:\s*+\‍[\s*+\d*+\s*+\‍])*+))\s*+(?:;|$))",
50 pcre2cpp::compile_options_bits::Multiline
51 );
52
53 /// @brief pattern to get struct data
54 static inline const auto structPattern =
55 pcre2cpp::regex(R"(^\s*+(?>(?>struct)\s++(?>(?<name>[a-zA-Z_]\w*+))\s*+(?>\{(?<body>[^}]*+)\})\s*+;)\s*+$)");
56
57 /// @brief pattern to get multiple structs
58 static inline const auto multiStructsPattern = pcre2cpp::regex(
59 R"(\s*+(?<struct>(?>struct|layout\s*+\‍([^)]*+\‍)\s*+(?>uniform|(?:\w++\s++)?buffer))\s++(?>(?:[a-zA-Z_]\w*+))\s*+\{(?>[^}]*+)\}\s*+(?>(?:\w*+\s*+));\s*+))",
60 pcre2cpp::compile_options_bits::Multiline
61 );
62 };

250 /// @brief returns ValueType based on type in string
251 static _GLSL_STRUCT_CONSTEXPR17 ValueType _get_value_type_from_string(const std::string_view type) noexcept {
252 // Option for vec and mat
253 if (type.empty() || type[0] == 'f') { return ValueType::Float; }
254
255 if (type[0] == 'i') { return ValueType::Int; }
256
257 if (type[0] == 'u') { return ValueType::Uint; }
258
259 if (type[0] == 'd') { return ValueType::Double; }
260
261 if (type[0] == 'b') { return ValueType::Bool; }
262
263 [[unlikely]] return ValueType::Int;
264 }

266 /// @brief adds variable of type T to struct or layout (if count != 0 then it adds array of variables of type T)
267 template<class T, class S>
268 static _GLSL_STRUCT_CONSTEXPR17 S& _add_variable(S& ref, const std::string_view name, const size_t count) {
269 if _GLSL_STRUCT_CONSTEXPR17 (std::is_same_v<S, _layout_type> || std::is_same_v<S, _struct_type>) {
270 if (count != 0) { ref.template add<std::vector<T> >(name, count); }
271 else { ref.template add<T>(name); }
272 }
273 return ref;
274 }

370 /// @brief extracts name and body of structure from string
371 _GLSL_STRUCT_CONSTEXPR20 std::pair<std::string, std::string> _extract_struct_data(const std::string_view structStr) {
372 pcre2cpp::match_result result;
373 if (_base_patterns::structPattern.match(structStr, result)) {
374 return std::make_pair(result.get_sub_result_value("name"), result.get_sub_result_value("body"));
375 }
376
377 if _GLSL_STRUCT_CONSTEXPR17 (has_ubo_pattern) {
378 if (_ubo_pattern::pattern.match(structStr, result)) {
379 return std::make_pair(result.get_sub_result_value("name"), result.get_sub_result_value("body"));
380 }
381 }
382
383 if _GLSL_STRUCT_CONSTEXPR17 (has_ssbo_pattern) {
384 if (_ssbo_pattern::pattern.match(structStr, result)) {
385 return std::make_pair(result.get_sub_result_value("name"), result.get_sub_result_value("body"));
386 }
387 }
388
389 glsl_struct_assert(false, "Provided string '{}' was not a struct definition", structStr);
390 return std::make_pair(std::string(), std::string());
391 }
392
393 /// @brief extracts struct strings from structs list
394 _GLSL_STRUCT_CONSTEXPR20 std::vector<std::string> _extract_structs(const std::string_view structsStr) {
395 std::vector<pcre2cpp::match_result> results;
396 if (!_base_patterns::multiStructsPattern.match_all(structsStr, results)) {
397 glsl_struct_assert(false, "Provided structs string '{}' didn't have any struct definition", structsStr);
398 return {};
399 }
400
401 std::vector<std::string> structs;
402 structs.reserve(results.size());
403 for (const auto& result : results) { structs.push_back(result.get_sub_result_value("struct")); }
404 return structs;
405 }

417 /// @brief loads layout from structsBodies to structsLayouts and returns if there is a layout with given name in structsLayouts
418 _GLSL_STRUCT_CONSTEXPR17 bool _load_layout(const std::string_view structName) {
419 if (!_contains_struct_layout(structName) && _contains_struct_body(structName)) {
420 _structsLayouts.emplace(structName.data(), _create_layout(_structsBodies.at(structName.data())));
421 _structsBodies.erase(structName.data());
422 }
423
424 return _contains_struct_layout(structName);
425 }

453 /// @brief adds variable of type given in string and with name and count
454 template<class S>
455 _GLSL_STRUCT_CONSTEXPR20 S& _add_variable(S& ref, const std::string_view type, const std::string_view name,
456 const size_t count = 0) {
457 pcre2cpp::match_result result;
458
459 // Check if scalar type
460 if (_base_patterns::scalarsPattern.match_at(type, result)) {
461 return _add_scalar_variable(ref, _get_value_type_from_string(result.get_sub_result_value("scalar")), name,
462 count);
463 }
464
465 // Check if vec type
466 if (_base_patterns::vecPattern.match_at(type, result)) {
467 size_t length;
468 mstd::strtounum(result.get_sub_result_value("length"), length);
469
470 return _add_vec_variable(ref, _get_value_type_from_string(result.get_sub_result_value("scalar")), length,
471 name, count);
472 }
473
474 // Check if mat type
475 if (_base_patterns::matPattern.match_at(type, result)) {
476 const ValueType valueType = _get_value_type_from_string(result.get_sub_result_value("scalar"));
477
478 size_t cols;
479 mstd::strtounum(result.get_sub_result_value("cols"), cols);
480
481 if (!result.has_sub_value("rows")) { return _add_mat_variable(ref, valueType, cols, cols, name, count); }
482
483 size_t rows;
484 mstd::strtounum(result.get_sub_result_value("rows"), rows);
485
486 return _add_mat_variable(ref, valueType, cols, rows, name, count);
487 }
488
489 // Check if struct type
490 if (_load_layout(type.data())) {
491 ref.add(name, _structsLayouts.at(type.data()));
492 return ref;
493 }
494
495 glsl_struct_assert(false, "Provided type str '{}' was not a correct variable type", type);
496 return ref;
497 }
498
499 /// @brief adds variable given in string
500 template<class S>
501 _GLSL_STRUCT_CONSTEXPR20 S& _add_variable(S& ref, const std::string_view varStr) {
502 pcre2cpp::match_result result;
503 if (!_base_patterns::variablePattern.match_at(varStr, result)) {
504 glsl_struct_assert(false, "Provided var str '{}' was not a correct variable", varStr);
505 return ref;
506 }
507
508 size_t count = 0;
509 if (result.has_sub_value("count")) { mstd::strtounum(result.get_sub_result_value("count"), count); }
510 else if (result.has_sub_value("array")) { count = 1; }
511
512 return _add_variable(ref, result.get_sub_result_value("type"), result.get_sub_result_value("name"), count);
513 }
514
515 /// @brief adds variables given in var list string
516 template<class S>
517 _GLSL_STRUCT_CONSTEXPR20 S& _add_variables(S& ref, const std::string_view varsStr) {
518 std::vector<pcre2cpp::match_result> results;
519 if (!_base_patterns::multiVariablesPattern.match_all(varsStr, results)) {
520 glsl_struct_assert(false, "Provided vars str '{}' was not a correct variables list", varsStr);
521 return ref;
522 }
523
524 for (const auto& result : results) { _add_variable(ref, result.get_sub_result_value("var")); }
525
526 return ref;
527 }
