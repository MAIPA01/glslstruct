55 /// @brief type visitor
56 template<class Type>
57 void operator()(Type&&) noexcept {
58 using type = std::decay_t<Type>;
59 _result = std::is_same_v<T, type>;

111 /**
112 * @brief requests is_of_type visitor to visit any glsl type
113 * @ingroup type_visitors
114 * @param baseType type which is required
115 */
116 #if _GLSL_STRUCT_HAS_CXX20
117 template<utils::glsl_type T>
118 #else
119 template<class T, std::enable_if_t<utils::is_glsl_type_v<T>, bool> = true>

121 [[nodiscard]] _GLSL_STRUCT_EXPORT _GLSL_STRUCT_INLINE17 _GLSL_STRUCT_CONSTEXPR20 bool is_of_type(const std::shared_ptr<T>&,
122 const BaseType baseType) {
123 switch (baseType) {
124 case BaseType::Scalar: return std::is_same_v<T, scalar_type>;
125 case BaseType::Vec: return std::is_same_v<T, vec_type>;
126 case BaseType::Mat: return std::is_same_v<T, mat_type>;
127 case BaseType::Struct: return std::is_same_v<T, struct_type>;
128 case BaseType::Array: return std::is_same_v<T, array_type>;
129 default: return false;
130 }
131 }
