56 /**
57 * @brief glsl type visitor concept
58 * @ingroup type_visitors

60 template<class T>
61 concept type_visitor = requires (T& visitor) {
62 { visitor(std::declval<const scalar_type&>()) } -> std::same_as<void>;
63 { visitor(std::declval<const vec_type&>()) } -> std::same_as<void>;
64 { visitor(std::declval<const mat_type&>()) } -> std::same_as<void>;
65 { visitor(std::declval<const struct_type&>()) } -> std::same_as<void>;
66 { visitor(std::declval<const array_type&>()) } -> std::same_as<void>;
67 };
68
69 template<class T>
70 static _GLSL_STRUCT_CONSTEXPR17 bool is_type_visitor_v = type_visitor<T>;

72 template<class T>
73 struct is_type_visitor : std::bool_constant<is_type_visitor_v<T> > {};
