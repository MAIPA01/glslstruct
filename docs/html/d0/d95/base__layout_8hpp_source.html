129 [[nodiscard]] static _GLSL_STRUCT_CONSTEXPR17 size_t _calculate_alignment_offset(const size_t currentOffset,
130 const size_t baseAlignment) noexcept {
131 // CALCULATE ALIGNMENT
132 const size_t alignmentOffset = ceil_to_nearest_multiple(currentOffset, baseAlignment);
133
134 // CHECK IF NOT OVERFLOWED
135 if (alignmentOffset < currentOffset) {
136 glsl_struct_assert(false, "Data overflow would happen!");
137 return bad_offset();
138 }
139
140 return alignmentOffset;
141 }

144 [[nodiscard]] static _GLSL_STRUCT_CONSTEXPR17 size_t _calculate_padding(const size_t currentOffset,
145 const size_t baseAlignment) noexcept {
146 return _calculate_alignment_offset(currentOffset, baseAlignment) - currentOffset;
147 }

195 [[nodiscard]] static _GLSL_STRUCT_CONSTEXPR17 size_t _add(size_t& currentOffset, const size_t baseAlignment,
196 const size_t baseOffset) noexcept {
197 // GET ALIGNMENT OFFSET
198 const size_t alignmentOffset = _calculate_alignment_offset(currentOffset, baseAlignment);
199
200 if (alignmentOffset == bad_offset()) { return alignmentOffset; }
201
202 // CHECK OFFSET BOUNDARIES
203 size_t tempOffset = alignmentOffset;
204 if (!_move_current_offset(tempOffset, baseOffset)) { return bad_offset(); }
205
206 // MOVE CURRENT OFFSET
207 currentOffset = tempOffset;
208 return alignmentOffset;
209 }

212 [[nodiscard]] static _GLSL_STRUCT_CONSTEXPR20 std::vector<size_t> _add_array(size_t& currentOffset,
213 const size_t arrayBaseAlignment, const size_t elemBaseOffset, const size_t count) noexcept {
214 // CALCULATE ARRAY ELEMENTS OFFSETS
215 size_t tempOffset = currentOffset;
216
217 std::vector<size_t> arrayOffsets;
218 arrayOffsets.reserve(count);
219 for (size_t i = 0, elemOffset; i < count; ++i) {
220 elemOffset = _add(tempOffset, arrayBaseAlignment, elemBaseOffset);
221
222 if (elemOffset == bad_offset()) { return std::vector<size_t>(); }
223
224 arrayOffsets.push_back(elemOffset);
225 }
226
227 // MOVE CURRENT OFFSET
228 currentOffset = tempOffset;
229 return arrayOffsets;
230 }

593 [[nodiscard]] _GLSL_STRUCT_CONSTEXPR20 std::vector<size_t> _add_scalar_array(const std::string_view name,
594 const ValueType valueType, const size_t count) {
595 // CHECK COUNT
596 if (!_is_array_count_good(count)) { return std::vector<size_t>(); }
597
598 // BEFORE ADD SCALAR ARRAY
599 _before_add_array();
600
601 // GET BASE OFFSET
602 const size_t baseOffset = get_value_type_size(valueType);
603
604 // GET ARRAY ALIGNMENT
605 const size_t arrayBaseAlignment = _get_scalar_array_alignment(valueType);
606
607 // GET ALIGNMENT OFFSETS
608 size_t tempOffset = _currentOffset;
609 const std::vector<size_t> alignmentOffsets = _add_array(tempOffset, arrayBaseAlignment, baseOffset, count);
610
611 // CHECK ADD
612 if (alignmentOffsets.empty()) { return alignmentOffsets; }
613
614 // GET ARRAY SIZE
615 const size_t arraySize = tempOffset - alignmentOffsets.front();
616
617 // GET ARRAY PADDING
618 const size_t arrayPadding = _calculate_padding(tempOffset, arrayBaseAlignment);
619
620 // APPLY PADDING TO CURRENT OFFSET
621 if (!_move_current_offset(tempOffset, arrayPadding)) { return std::vector<size_t>(); }
622 _currentOffset = tempOffset;
623
624 // ADD VARIABLE DATA
625 _add_array_variable(name, alignmentOffsets, true,
626 #if _GLSL_STRUCT_HAS_TYPES
627 count, std::make_shared<scalar_type>(valueType, baseOffset),
628 #else
629 baseOffset,
630 #endif
631 arraySize)
632 .set_padding(arrayPadding);
633
634 // AFTER ADD SCALAR ARRAY
635 _after_add_array(arraySize, arrayBaseAlignment);
636
637 return alignmentOffsets;
638 }

778 [[nodiscard]] _GLSL_STRUCT_CONSTEXPR20 std::vector<std::vector<size_t> > _add_mat_array(const std::string_view name,
779 const size_t columns, const size_t rows, const ValueType valueType, const size_t count) {
780 // CHECK COUNT
781 if (!_is_array_count_good(count)) { return std::vector<std::vector<size_t> >(); }
782
783 // CHECK MAT COLUMNS
784 if (!_is_vec_length_good(columns)) { return std::vector<std::vector<size_t> >(); }
785
786 // CHECK MAT ROWS
787 if (!_is_vec_length_good(rows)) { return std::vector<std::vector<size_t> >(); }
788
789 // BEFORE ADD MAT ARRAY
790 _before_add_array();
791
792 // GET BASE OFFSET
793 const size_t vecBaseOffset = get_value_type_size(valueType) * rows;
794
795 // GET ARRAY ALIGNMENT
796 const size_t matBaseAlignment = _get_mat_alignment(valueType, rows);
797
798 // GET MATS ALIGNMENT OFFSETS
799 size_t tempOffset = _currentOffset;
800
801 size_t matSize = 0;
802 size_t matPadding = 0;
803
804 std::vector<std::vector<size_t> > alignmentOffsetsPerMat;
805 alignmentOffsetsPerMat.reserve(count);
806 for (size_t i = 0; i < count; ++i) {
807 // GET VEC ALIGNMENT OFFSETS
808 const std::vector<size_t> alignmentOffsets = _add_array(tempOffset, matBaseAlignment, vecBaseOffset, columns);
809
810 // CHECK ADD
811 if (alignmentOffsets.empty()) { return std::vector<std::vector<size_t> >(); }
812
813 // CALCULATE SIZE AND PADDING ONCE
814 if (matSize == 0) {
815 // CALCULATE SIZE
816 matSize = tempOffset - alignmentOffsets.front();
817
818 // CALCULATE PADDING
819 matPadding = _calculate_padding(tempOffset, matBaseAlignment);
820 }
821
822 // APPLY PADDING TO CURRENT OFFSET
823 if (i != count - 1) {
824 if (!_move_current_offset(tempOffset, matPadding)) { return std::vector<std::vector<size_t> >(); }
825 }
826
827 alignmentOffsetsPerMat.push_back(alignmentOffsets);
828 }
829
830 // GET ARRAY SIZE
831 const size_t arraySize = tempOffset - alignmentOffsetsPerMat.front().front();
832
833 // APPLY PADDING TO CURRENT OFFSET
834 if (!_move_current_offset(tempOffset, matPadding)) { return std::vector<std::vector<size_t> >(); }
835 _currentOffset = tempOffset;
836
837 // ADD VARIABLE DATA
838 _add_mat_array_variable(name, alignmentOffsetsPerMat, true, count,
839 #if _GLSL_STRUCT_HAS_TYPES
840 valueType, columns, rows,
841 #endif
842 vecBaseOffset, matSize, arraySize, matPadding);
843
844 // AFTER ADD SCALAR ARRAY
845 _after_add_array(arraySize, matBaseAlignment);
846
847 return alignmentOffsetsPerMat;
848 }

852 const size_t baseOffset, const mstd::ordered_map<std::string, var_data>& values) {
853 // CHECK VALUES

858
859 // GET ALIGNMENT OFFSET

861
862 // CHECK ADD

864
865 // ADD VARIABLE DATA
866 _add_variable(
867 name, alignmentOffset, true,
868 #if _GLSL_STRUCT_HAS_TYPES
869 std::make_shared<struct_type>(values, baseOffset)
870 #else
871 baseOffset
872 #endif
873 );
874

883 #endif

890
891 return alignmentOffset;
892 }
